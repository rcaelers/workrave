cmake_minimum_required(VERSION 3.28)
project(Workrave LANGUAGES NONE)

# Configure the existing distribution rules around Linux-built binaries.
# No application or dependency compiler is used in this project.
foreach(required WORKRAVE_SOURCE_DIR WORKRAVE_SDK_DIR)
  if(NOT DEFINED ${required})
    message(FATAL_ERROR "${required} is required")
  endif()
endforeach()
file(READ "${CMAKE_BINARY_DIR}/handoff.json" handoff)
string(JSON WORKRAVE_VERSION GET "${handoff}" package version)
string(JSON WORKRAVE_CHANNEL GET "${handoff}" package channel)
set(BINDIR bin)
set(BINDIR32 bin32)
set(HAVE_CRASHPAD "${WITH_CRASHPAD}")
set(HAVE_SBOM "${WITH_SBOM}")
set(SYS_ROOT "$ENV{LLVM_MINGW_ROOT}/x86_64-w64-mingw32")
find_program(CMAKE_OBJDUMP NAMES llvm-objdump REQUIRED)
install(CODE "set(CMAKE_OBJDUMP \"${CMAKE_OBJDUMP}\")")
file(GLOB sdk_packages LIST_DIRECTORIES TRUE "${WORKRAVE_SDK_DIR}/packages/*")
set(CMAKE_PREFIX_PATH ${sdk_packages} "${WORKRAVE_SDK_DIR}/runtime/..")
set(QT_BIN_DIR "${WORKRAVE_SDK_DIR}/packages/qt/bin")
set(QT_QTPATHS_EXECUTABLE "${QT_BIN_DIR}/qtpaths.exe")
set(WINDEPLOYQT_EXECUTABLE "${QT_BIN_DIR}/windeployqt.exe")
if(WITH_SIGN)
  if(NOT EXISTS "${WITH_SIGN_TOOL}")
    message(FATAL_ERROR "WITH_SIGN_TOOL must point to the native Ship executable")
  endif()
  set(SIGN_TOOL "${WITH_SIGN_TOOL}")
  file(TO_NATIVE_PATH "${SIGN_TOOL}" SIGN_TOOL_WIN)
  set(SIGN_TOOL_ISCC "$q${SIGN_TOOL_WIN}$q sign authenticode $f")
endif()
add_custom_target(workrave)
add_subdirectory("${WORKRAVE_SOURCE_DIR}/ui/app/toolkits/qt/dist/windows" distribution)
